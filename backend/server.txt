const express = require("express");
const cors = require("cors");
const sqlite3 = require("sqlite3").verbose();

const app = express();
const PORT = 5000;

app.use(cors());
app.use(express.json());


// ================= DATABASE =================

const db = new sqlite3.Database("./grievance.db");

db.run(`
    CREATE TABLE IF NOT EXISTS grievances (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        ticket_id TEXT UNIQUE,
        name TEXT,
        phone TEXT,
        email TEXT,
        complaint TEXT,
        category TEXT,
        priority TEXT,
        department TEXT,
        status TEXT DEFAULT 'Pending',
        eta TEXT,
        created_at TEXT
    )
`);


// ================= AI ANALYSIS =================

function analyzeComplaint(complaint) {

    const text = complaint.toLowerCase();

    let category = "Municipal";
    let department = "Municipal Corporation";
    let priority = "Low";

    // Category
    if (
        text.includes("water") ||
        text.includes("pipe") ||
        text.includes("leakage") ||
        text.includes("tap")
    ) {
        category = "Water";
        department = "Water Supply Department";
    }

    else if (
        text.includes("electricity") ||
        text.includes("power") ||
        text.includes("light") ||
        text.includes("transformer")
    ) {
        category = "Electricity";
        department = "Electricity Board";
    }

    else if (
        text.includes("road") ||
        text.includes("pothole") ||
        text.includes("bridge") ||
        text.includes("street")
    ) {
        category = "Roads";
        department = "Public Works Department";
    }

    else if (
        text.includes("hospital") ||
        text.includes("doctor") ||
        text.includes("medicine") ||
        text.includes("ambulance")
    ) {
        category = "Healthcare";
        department = "Health Department";
    }

    else if (
        text.includes("school") ||
        text.includes("college") ||
        text.includes("teacher") ||
        text.includes("exam")
    ) {
        category = "Education";
        department = "Education Department";
    }


    // Priority

    const highWords = [
        "emergency",
        "urgent",
        "danger",
        "accident",
        "fire",
        "critical",
        "life threat"
    ];

    const mediumWords = [
        "serious",
        "blocked",
        "not working",
        "immediately"
    ];

    if (highWords.some(word => text.includes(word))) {
        priority = "High";
    }

    else if (mediumWords.some(word => text.includes(word))) {
        priority = "Medium";
    }


    return {
        category,
        department,
        priority
    };
}

// ================= REPORT GRIEVANCE =================

app.post("/api/grievances", (req, res) => {

    const {
        name,
        phone,
        email,
        complaint
    } = req.body;


    if (!complaint) {

        return res.status(400).json({
            message: "Complaint is required"
        });

    }


    // AI analysis
    const ai = analyzeComplaint(complaint);


    // Ticket ID
    const ticketId =
        "GA" +
        Math.floor(100000 + Math.random() * 900000);


    // ETA
    let days = 5;

    if (ai.priority === "High") {
        days = 1;
    }

    else if (ai.priority === "Medium") {
        days = 3;
    }


    const eta = new Date();

    eta.setDate(
        eta.getDate() + days
    );


    const etaDate =
        eta.toISOString().split("T")[0];


    const createdAt =
        new Date().toISOString();


    const sql = `
        INSERT INTO grievances
        (
            ticket_id,
            name,
            phone,
            email,
            complaint,
            category,
            priority,
            department,
            status,
            eta,
            created_at
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `;


    db.run(
        sql,
        [
            ticketId,
            name || "",
            phone || "",
            email || "",
            complaint,
            ai.category,
            ai.priority,
            ai.department,
            "Pending",
            etaDate,
            createdAt
        ],

        function (error) {

            if (error) {

                return res.status(500).json({
                    message: error.message
                });

            }


            res.json({

                success: true,

                message:
                    "Grievance submitted successfully",

                ticket_id: ticketId,

                category: ai.category,

                priority: ai.priority,

                department: ai.department,

                status: "Pending",

                eta: etaDate

            });

        }
    );

});


// ================= TRACK COMPLAINT =================

app.get("/api/grievances/:ticketId", (req, res) => {

    const ticketId = req.params.ticketId;


    db.get(
        `
        SELECT *
        FROM grievances
        WHERE ticket_id = ?
        `,
        [ticketId],

        (error, row) => {

            if (error) {

                return res.status(500).json({
                    message: error.message
                });

            }


            if (!row) {

                return res.status(404).json({
                    message: "Ticket not found"
                });

            }


            res.json({
                success: true,
                grievance: row
            });

        }
    );

});

// ================= DASHBOARD =================

app.get("/api/dashboard", (req, res) => {

    const queries = {

        total: `
            SELECT COUNT(*) AS count
            FROM grievances
        `,

        resolved: `
            SELECT COUNT(*) AS count
            FROM grievances
            WHERE status = 'Resolved'
        `,

        pending: `
            SELECT COUNT(*) AS count
            FROM grievances
            WHERE status = 'Pending'
        `,

        progress: `
            SELECT COUNT(*) AS count
            FROM grievances
            WHERE status = 'In Progress'
        `

    };


    db.get(queries.total, [], (e, total) => {

        db.get(
            queries.resolved,
            [],
            (e, resolved) => {

                db.get(
                    queries.pending,
                    [],
                    (e, pending) => {

                        db.get(
                            queries.progress,
                            [],
                            (e, progress) => {

                                const successRate =
                                    total.count === 0
                                        ? 0
                                        : Math.round(
                                            (resolved.count /
                                                total.count) * 100
                                        );


                                res.json({

                                    total: total.count,

                                    resolved:
                                        resolved.count,

                                    pending:
                                        pending.count,

                                    in_progress:
                                        progress.count,

                                    success_rate:
                                        successRate

                                });

                            }
                        );

                    }
                );

            }
        );

    });

});
