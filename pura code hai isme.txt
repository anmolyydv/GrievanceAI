let currentSlide = 0;

const slideImage = document.getElementById("slideImage");
const dots = document.querySelectorAll(".dot");


function showSlide(index) {

    currentSlide = index;

    slideImage.src = images[currentSlide];

    dots.forEach((dot, i) => {

        dot.classList.toggle(
            "active",
            i === currentSlide
        );

    });

}


function nextSlide() {

    currentSlide++;

    if (currentSlide >= images.length) {
        currentSlide = 0;
    }

    showSlide(currentSlide);
}


function prevSlide() {

    currentSlide--;

    if (currentSlide < 0) {
        currentSlide = images.length - 1;
    }

    showSlide(currentSlide);
}


// Automatic slide every 4 seconds

setInterval(nextSlide, 4000);

def parse_dt(value):

def guess_language(text):
    if re.search(r"[\u0900-\u097F]", text):
        return "Hindi"
    if re.search(r"[\u0980-\u09FF]", text):
        return "Bengali"
    if re.search(r"[\u0A80-\u0AFF]", text):
        return "Gujarati"
    if re.search(r"[\u0B80-\u0BFF]", text):
        return "Tamil"
    if re.search(r"[\u0C00-\u0C7F]", text):
        return "Telugu"
    if re.search(r"[\u0D00-\u0D7F]", text):
        return "Malayalam"
    if re.search(r"[\u0600-\u06FF]", text):
        return "Urdu/Arabic"
    return "English/Hinglish"

def save_image(file):
    if not file or not file.filename:
        return None

    if not file.content_type or not file.content_type.startswith("image/"):
        raise ValueError("Only image files are allowed.")

    ext = os.path.splitext(file.filename)[1].lower() or ".jpg"
    name = f"{uuid.uuid4().hex}{ext}"
    return os.path.join(UPLOAD_DIR, name)

async def store_upload(file):
    path = save_image(file)

    if not path:
        return None

    data = await file.read()

    with open(path, "wb") as f:
        f.write(data)

    return path

def openai_ai(text, image_path=None):
    if not OPENAI_API_KEY or OpenAI is None:

    client = OpenAI(api_key=OPENAI_API_KEY)

    content = [

            "text": instruction
        }
    ]

    if image_path:
        with open(image_path, "rb") as f:
            encoded = base64.b64encode(f.read()).decode("utf-8")

        mime = "image/jpeg"
        ext = os.path.splitext(image_path)[1].lower()

        if ext == ".png":
            mime = "image/png"
        elif ext == ".webp":
            mime = "image/webp"

        content.append({
            "type": "input_image",
            "image_url": f"data:{mime};base64,{encoded}"
        })

    response = client.responses.create(
        model=AI_MODEL,
        input=[
            {
                "role": "user",
                "content": content
            }
        ]
    )

    raw = response.output_text.strip()

    raw = re.sub(r"^```(?:json)?", "", raw).strip()
    raw = re.sub(r"```$", "", raw).strip()

    try:
        result = json.loads(raw)
    except Exception:
        return fallback_ai(text)

    required = [
        "language",
        "problem",
        "severity",
        "priority",
        "department",
        "keywords",
        "summary",
        "confidence"
    ]

    if not all(k in result for k in required):
        return fallback_ai(text)

    return result

def get_location(lat, lon):
    if lat is None or lon is None:
        return {
            "area": "",
            "address": ""
        }

    try:
        response = requests.get(
            "https://nominatim.openstreetmap.org/reverse",
            params={
                "lat": lat,
                "lon": lon,
                "format": "json",
                "zoom": 18,
                "addressdetails": 1
            },
            headers={
                "User-Agent": "CitizenComplaintAI/1.0"
            },
            timeout=10
        )

        data = response.json()
        address = data.get("address", {})

        area = (
            address.get("neighbourhood")
            or address.get("suburb")
            or address.get("town")
            or address.get("city")
            or ""
        )

        return {
            "area": area,
            "address": data.get("display_name", "")
        }

    except Exception:
        return {
            "area": "",
            "address": ""
        }

def distance_km(lat1, lon1, lat2, lon2):
    if None in (lat1, lon1, lat2, lon2):
        return None

    radius = 6371

    p1 = math.radians(lat1)
    p2 = math.radians(lat2)

    dp = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)

    a = (
        math.sin(dp / 2) ** 2
        + math.cos(p1)
        * math.cos(p2)
        * math.sin(dl / 2) ** 2
    )

    return radius * 2 * math.atan2(
        math.sqrt(a),
        math.sqrt(1 - a)
    )

def similar_complaints(text, lat=None, lon=None):
    conn = db()

    rows = conn.execute(
        "SELECT * FROM complaints WHERE status != 'Resolved'"
    ).fetchall()

    conn.close()

    if not rows:
        return []

    corpus = [text] + [
        row["complaint_text"]
        for row in rows
    ]

    vectorizer = TfidfVectorizer(
        analyzer="char_wb",
        ngram_range=(3, 5),
        min_df=1
    )

    matrix = vectorizer.fit_transform(corpus)

    scores = cosine_similarity(
        matrix[0:1],
        matrix[1:]
    )[0]

    results = []

    for i, row in enumerate(rows):
        score = float(scores[i])

        distance = distance_km(
            lat,
            lon,
            row["latitude"],
            row["longitude"]
        )

        location_bonus = 0.12 if (
            distance is not None and
            distance <= 1.0
        ) else 0

        final_score = min(
            1.0,
            score + location_bonus
        )

        if final_score >= 0.42:
            results.append({
                "id": row["id"],
                "problem": row["problem"],
                "priority": row["priority"],
                "status": row["status"],
                "similarity": round(
                    final_score * 100,
                    1
                ),
                "distance_km": round(
                    distance,
                    2
                ) if distance is not None else None
            })

    results.sort(
        key=lambda x: x["similarity"],
        reverse=True
    )

    return results[:5]

def serialize(row):
    now = utc_now()

    deadline = (
        parse_dt(row["last_action_at"])
        + timedelta(hours=48)
    )

    remaining = max(
        0,
        int(
            (
                deadline - now
            ).total_seconds()
        )
    )

    hours = remaining // 3600
    minutes = (remaining % 3600) // 60

    return {
        "id": row["id"],
        "complaint": row["complaint_text"],
        "language": row["language"],
        "problem": row["problem"],
        "severity": row["severity"],
        "priority": row["priority"],
        "department": row["department"],
        "area": row["area"],
        "address": row["address"],
        "status": row["status"],
        "submitted_at": row["submitted_at"],
        "last_action_at": row["last_action_at"],
        "hours_remaining": hours,
        "minutes_remaining": minutes,
        "escalated": bool(row["escalated"]),
        "resolved_at": row["resolved_at"]
    }

async def check_and_escalate():
    while True:
        try:
            conn = db()

            rows = conn.execute(
                """
                SELECT *
                FROM complaints
                WHERE status != 'Resolved'
                AND escalated = 0
                """
            ).fetchall()

            now = utc_now()

            for row in rows:
                last_action = parse_dt(
                    row["last_action_at"]
                )

                if now - last_action >= timedelta(hours=48):
                    conn.execute(
                        """
                        UPDATE complaints
                        SET
                            escalated = 1,
                            escalated_at = ?,
                            status = 'Escalated to HOD'
                        WHERE id = ?
                        """,
                        (
                            now.isoformat(),
                            row["id"]
                        )
                    )

            conn.commit()
            conn.close()

        except Exception:
            pass

        await asyncio.sleep(60)
